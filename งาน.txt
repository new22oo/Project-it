-- 1. สร้างฐานข้อมูลชื่อ it_repair_system
CREATE DATABASE IF NOT EXISTS it_repair_system;

-- 2. สั่งให้ระบบเข้าไปใช้งานฐานข้อมูลนี้
USE it_repair_system;

-- 3. สร้างตารางผู้ใช้งาน (users)
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    password VARCHAR(100) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role VARCHAR(20) NOT NULL -- (user  technician  admin)
);

-- 4. สร้างตารางอุปกรณ์ IT (equipment)
CREATE TABLE equipment (
    equipment_id INT AUTO_INCREMENT PRIMARY KEY,
    equipment_name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    location VARCHAR(100) NOT NULL,
    status VARCHAR(20) NOT NULL
);

-- 5. สร้างตารางแจ้งซ่อม (repair_request)
CREATE TABLE repair_request (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    equipment_id INT,
    problem_detail TEXT,
    request_date DATE,
    status VARCHAR(20),
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (equipment_id) REFERENCES equipment(equipment_id)
);

-- 6. สร้างตารางบันทึกการซ่อม (repair_log)
CREATE TABLE repair_log (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    request_id INT,
    technician_id INT,
    action_detail TEXT,
    update_date DATE,
    FOREIGN KEY (request_id) REFERENCES repair_request(request_id),
    FOREIGN KEY (technician_id) REFERENCES users(user_id)
);
